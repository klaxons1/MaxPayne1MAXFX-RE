[ac_fan_loop]  				// The flapping of an old AC fan going round and round 
{
	[Properties] ID = ID_LOOPING; 
	[Default]
	{
		[Sound]
        Filename = ambient\ac\ac_fan_loop.wav;
		Volume = 0.8;
		Pitch = 22050;
		Random = 0%;  
        3DSound = YES;
        Looping = YES;
		Streamed = NO;
		Hotspot = ID_FAN_HOTSPOT;
		FallOff = ID_FAN_FALLOFF;
		Priority = ID_FAN_PRIORITY;
		ItemsToCache = 1;
		FrequencyScaling = TRUE;
	}
}
